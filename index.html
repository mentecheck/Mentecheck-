function mostrarResultado(){ document.getElementById('questionario').style.display='none'; document.getElementById('resultado-bloqueado').style.display='block'; const maxPontos=34; const pct=pontuacao/maxPontos; let zona,titulo,texto,classeZona; if(pct<0.33){ zona='🟢 Zona Verde'; titulo='Sua saúde mental está equilibrada'; texto='Seus indicadores estão dentro da faixa saudável.'; classeZona='zona-verde'; }else if(pct<0.66){ zona='🟡 Zona Amarela'; titulo='Sinais de desgaste identificados'; texto='Você apresenta alguns indicadores que merecem atenção.'; classeZona='zona-amarelo'; }else{ zona='🔴 Zona Vermelha'; titulo='Sinais importantes de sofrimento'; texto='Seus indicadores apontam para um nível significativo.'; classeZona='zona-vermelho'; } const badge=document.getElementById('zonaBadge'); badge.textContent=zona; badge.className=`zona-badge ${classeZona}`; document.getElementById('resultadoTitulo').textContent=titulo; document.getElementById('resultadoTexto').textContent=texto; localStorage.setItem('resultadoMenteCheck', JSON.stringify({ pontuacao: pontuacao, zona: zona, data: new Date().toLocaleString('pt-BR') })); window.scrollTo(0,0); }
